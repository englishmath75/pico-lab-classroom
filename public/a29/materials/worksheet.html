3. 초음파센서 핵심 이해

  • TRIG(D7): 초음파를 보내기 위한 신호를 출력합니다.
  • ECHO(D8): 초음파가 돌아오는 데 걸린 시간을 읽습니다.
  • 거리 계산 코드는 제공되며 학생은 계산된 distance를 조건에 사용합니다.
  • 유효한 거리값이 20cm 미만이면 부저가 울리고, 20cm 이상이면 멈춥니다.
알아두기 unsigned long duration은 초음파 왕복 시간을 저장하는 교사 제공 부분입니다. 현재 평가에서는 이 자료형 자체를 암기하거나 작성하는 것이 핵심 평가 대상이 아닙니다.

학생용 12개 빈칸 코드 읽기

실제 평가에서 제공되는 코드 구조입니다. 코드 내용은 수행평가 연습 페이지와 동일합니다.

// A29-UNO-80-v1.1 | Arduino Uno R3 | Student starter
// Replace __01__ ... __12__. This file will not compile before completion.
// Pins: TRIG D7, ECHO D8, LED D9, passive piezo D2, CDS A0, POT A1.
const int LIGHT_LIMIT = 400;

void setup() {
  pinMode(7, OUTPUT);
  pinMode(8, INPUT);
  pinMode(9, OUTPUT);
  pinMode(2, OUTPUT);
  digitalWrite(7, LOW);
  analogWrite(9, 0);
  noTone(2);
  Serial.begin(9600);
}

void loop() {
  int light = __01__(A0);
  int pot = __02__(A1);
  int brightness = map(pot, 0, 1023, __03__, __04__);

  // Teacher-provided ultrasonic measurement.
  digitalWrite(7, LOW);
  delayMicroseconds(2);
  digitalWrite(7, HIGH);
  delayMicroseconds(10);
  digitalWrite(7, LOW);
  unsigned long duration = pulseIn(8, HIGH, 30000UL);
  float distance = duration * 0.034f / 2.0f;

  int ledPwm = 0;
  int alarm = 0;
  // A timeout must not be treated as a distance of zero centimetres.
  if (duration > 0) {
    if (__05__) {
      alarm = 1;
      if (__06__) {
        ledPwm = __07__;
      }
    }
  }

  analogWrite(9, __08__);
  if (alarm == 1) {
    __09__(2, 1000);
  } else {
    __10__(2);
  }

  Serial.print("DIST_CM=");
  if (duration > 0) {
    Serial.print(__11__);
  } else {
    Serial.print("N/A");
  }
  Serial.print(" LIGHT_RAW="); Serial.print(__12__);
  Serial.print(" POT_RAW="); Serial.print(pot);
  Serial.print(" LED_PWM_CMD="); Serial.print(ledPwm);
  Serial.print(" BUZZER_CMD="); Serial.println(alarm);
  delay(200);
}
