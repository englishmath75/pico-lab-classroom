빨간 LED (+), 저항 직렬D9
초록 LED (+), 저항 직렬D10

LED 각각에 220~330Ω 직렬 저항을 사용하고 GND를 공통 연결하세요. CDS에는 10kΩ 저항을 사용해 전압분배 회로를 구성하세요. 예시 코드는 CDS를 5V 쪽, 10kΩ 저항을 GND 쪽에 연결한 기준입니다.

1단계 · 초음파 + 부저

20cm 미만이면 부저가 울립니다.

const int trig=7, echo=8, buz=2;
void setup(){
 pinMode(trig,OUTPUT);
 pinMode(echo,INPUT);
 pinMode(buz,OUTPUT);
 Serial.begin(9600);
}
void loop(){
 digitalWrite(trig,LOW);
 delayMicroseconds(2);
 digitalWrite(trig,HIGH);
 delayMicroseconds(10);
 digitalWrite(trig,LOW);
 long t=pulseIn(echo,HIGH,30000);
 float cm=t*0.0343/2;
 if(t>0 && cm<20) tone(buz,1000);
 else noTone(buz);
 Serial.println(cm);
 delay(200);
}

2단계 · 조도 + 가변저항

어두우면 빨간 LED가 켜지고, 가변저항으로 초록 LED 밝기를 조절합니다.

const int cds=A0, pot=A1, red=9, green=10;
void setup(){
 pinMode(red,OUTPUT);
 pinMode(green,OUTPUT);
 Serial.begin(9600);
}
void loop(){
 int light=analogRead(cds);
 int knob=analogRead(pot);
 if(light<400) digitalWrite(red,HIGH);
 else digitalWrite(red,LOW);
 analogWrite(green,map(knob,0,1023,0,255));
 Serial.print(light);
 Serial.print(" ");
 Serial.println(knob);
 delay(200);
}

3단계 · 전체 통합

세 가지 출력이 독립적으로 작동합니다.

const int trig=7, echo=8, buz=2;
const int cds=A0, pot=A1, red=9, green=10;
void setup(){
 pinMode(trig,OUTPUT);
 pinMode(echo,INPUT);
 pinMode(buz,OUTPUT);
 pinMode(red,OUTPUT);
 pinMode(green,OUTPUT);
 Serial.begin(9600);
}
void loop(){
 digitalWrite(trig,LOW);
 delayMicroseconds(2);
 digitalWrite(trig,HIGH);
 delayMicroseconds(10);
 digitalWrite(trig,LOW);
 long t=pulseIn(echo,HIGH,30000);
 float cm=t*0.0343/2;
 if(t>0 && cm<20) tone(buz,1000);
 else noTone(buz);
 int light=analogRead(cds);
 if(light<400) digitalWrite(red,HIGH);
 else digitalWrite(red,LOW);
 int knob=analogRead(pot);
 analogWrite(green,map(knob,0,1023,0,255));
 Serial.print("cm="); Serial.print(cm);
 Serial.print(" light="); Serial.print(light);
 Serial.print(" pot="); Serial.println(knob);
 delay(200);
}

검증 순서

  1. 초음파 앞에 물체를 20cm보다 가까이 대면 부저가 울리는지 확인합니다.
  2. CDS를 손으로 가리면 빨간 LED가 켜지는지 확인합니다. 주변 조도에 따라 기준값 400은 조정할 수 있습니다.
  3. 가변저항을 돌리면 초록 LED 밝기가 변하는지 확인합니다.
