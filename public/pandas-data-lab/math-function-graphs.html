Turtle .py 다운로드
# math 모듈과 반복문을 사용해 여러 수학 함수를 그리는 코드

import math
import turtle

screen = turtle.Screen()
screen.bgcolor("black")

pen = turtle.Turtle()
pen.speed(0)
pen.hideturtle()
pen.pensize(2)

# 좌표계 설정
pen.penup()
pen.goto(-300, 0)
pen.pendown()
pen.color("white")
pen.goto(300, 0)

pen.penup()
pen.goto(0, -300)
pen.pendown()
pen.goto(0, 300)

# 다항함수: y = ax^2 + b
pen.penup()
pen.color("cyan")
a, b = 0.01, -50
for x in range(-200, 201):
    y = a * (x**2) + b
    if abs(y) <= 300:
        pen.goto(x, y)
        pen.pendown()
    else:
        pen.penup()

# 지수함수: y = ae^x + b (스케일 조정)
pen.penup()
pen.color("magenta")
a, b = 5, -100
for x in range(-150, 60):
    y = a * math.exp(x / 20) + b
    if abs(y) <= 300:
        pen.goto(x, y)
        pen.pendown()
    else:
        pen.penup()

# 로그함수: y = aln(x) + b (스케일 조정)
pen.penup()
pen.color("yellow")
a, b = 50, 0
for x in range(1, 300):
    y = a * math.log(x / 10) + b
    if abs(y) <= 300:
        pen.goto(x, y)
        pen.pendown()
    else:
        pen.penup()

# 원의 방정식: x^2 + y^2 = r^2
pen.penup()
pen.color("lime")
r = 100
for angle in range(361):
    rad = math.radians(angle)
    x = r * math.cos(rad)
    y = r * math.sin(rad)
    pen.goto(x, y)
    if angle == 0:
        pen.pendown()

turtle.done()

실습 1. 일차함수와 이차함수 (두 그래프 창)

먼저 x값 200개를 만든 뒤, 첫 번째에는 y = 2*x + 4, 두 번째에는 y = x**2를 계산합니다. 첫 번째 그래프 창을 닫으면 두 번째 그래프가 표시됩니다. 코드가 끝까지 실행되는지 확인하세요.

① 일차·이차함수 .py 다운로드
import numpy as np
import matplotlib.pyplot as plt

# 1. x 구간 설정 (-10부터 10까지 200개 점)
x = np.linspace(-10, 10, 200)

# 2. y = 2*x + 4 그리기
y = 2 * x + 4
plt.plot(x, y, color='cyan', label='y = 2x + 4')
plt.axhline(0, color='gray', linewidth=1)  # x축
plt.axvline(0, color='gray', linewidth=1)  # y축
plt.grid(True, linestyle='--', alpha=0.6)
plt.title('y = 2x + 4 Graph')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()
plt.show()

# 3. y = x^2 그리기
# 첫 번째 창을 닫으면 두 번째 그래프가 나옵니다.
y = x**2
plt.plot(x, y, color='cyan', label='y = x^2')
plt.axhline(0, color='gray', linewidth=1)  # x축
plt.axvline(0, color='gray', linewidth=1)  # y축
plt.grid(True, linestyle='--', alpha=0.6)
plt.title('y = x^2 Graph')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()
plt.show()
코드 읽기: np.linspace(-10, 10, 200)은 -10부터 10까지 200개의 x값을 만듭니다. x**2의 **는 제곱을 뜻합니다. plt.axhline(0)은 가로 x축, plt.axvline(0)은 세로 y축을 표시합니다.

실습 2. 사인함수와 코사인함수 (한 그래프에 함께)

이번에는 -2π부터 2π까지 400개의 x값을 만들고 np.sin(x), np.cos(x)를 계산합니다. plt.plot()을 두 번 실행하면 하나의 그래프 창에 사인과 코사인이 함께 나타납니다.

② 사인·코사인 .py 다운로드
import numpy as np
import matplotlib.pyplot as plt

# 1. x 구간 설정 (-2π부터 2π까지 400개 점 생성)
x = np.linspace(-2 * np.pi, 2 * np.pi, 400)

# 2. y값 계산 (사인, 코사인)
y_sin = np.sin(x)
y_cos = np.cos(x)

# 3. 도화지 생성 및 그래프 그리기
plt.plot(x, y_sin, label='sin(x)')
plt.plot(x, y_cos, label='cos(x)')

# 4. 좌표축 및 눈금 설정
plt.axhline(0, color='gray', linewidth=1)  # x축
plt.axvline(0, color='gray', linewidth=1)  # y축
plt.grid(True, linestyle=':', alpha=0.6)

# x축 눈금을 π(파이) 단위 라벨로 보기 쉽게 표시
plt.xticks(
    [-2 * np.pi, -np.pi, 0, np.pi, 2 * np.pi],
    ['-2π', '-π', '0', 'π', '2π']
)
plt.ylim(-1.5, 1.5)  # y축 범위 고정

# 5. 제목 및 범례 표시
plt.title('(Sine & Cosine)')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()

# 6. 화면 출력
plt.show()
수업에서 꼭 알려주세요. plt.legend()는 그래프에 label을 지정하고 plt.plot()으로 그린 다음에 실행해야 범례가 표시됩니다. 학생이 제시한 코드를 바탕으로 이 실행 순서만 바로잡았습니다. 사인·코사인은 라디안을 입력하며, plt.xticks()는 x축 눈금을 −2π, −π, 0, π, 2π로 보여 줍니다.
